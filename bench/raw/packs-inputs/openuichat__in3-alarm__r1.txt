root = Card([header, form])
header = CardHeader("Alarm Setup", "Configure your alarm settings")
form = Form("alarm-setup", formButtons, [fcTime, fcDays, fcSound])
formButtons = Buttons([btnSave, btnCancel], "row")
btnSave = Button("Save Alarm", { type: "continue_conversation", context: "Save the alarm with the configured settings" }, "primary")
btnCancel = Button("Cancel", { type: "continue_conversation", context: "Cancel alarm setup" }, "secondary")
fcTime = FormControl("Alarm Time", timeInput, "Set the alarm time (24-hour format)")
timeInput = Input("alarm-time", "07:30", "text", { required: true, pattern: "^([01]?[0-9]|2[0-3]):[0-5][0-9]$" })
fcDays = FormControl("Repeat Days", daysChips, "Select which days the alarm repeats")
daysChips = Chips("repeat-days", "multiple", [dayMon, dayTue, dayWed, dayThu, dayFri, daySat, daySun], { required: true }, ["mon", "tue", "wed", "thu", "fri"])
dayMon = ChipItem("mon", "Mon", dayMonIcon)
dayMonIcon = Icon("sun", "nature")
dayTue = ChipItem("tue", "Tue", dayTueIcon)
dayTueIcon = Icon("sun", "nature")
dayWed = ChipItem("wed", "Wed", dayWedIcon)
dayWedIcon = Icon("sun", "nature")
dayThu = ChipItem("thu", "Thu", dayThuIcon)
dayThuIcon = Icon("sun", "nature")
dayFri = ChipItem("fri", "Fri", dayFriIcon)
dayFriIcon = Icon("sun", "nature")
daySat = ChipItem("sat", "Sat", daySatIcon)
daySatIcon = Icon("sun", "nature")
daySun = ChipItem("sun", "Sun", daySunIcon)
daySunIcon = Icon("sun", "nature")
fcSound = FormControl("Alarm Sound", soundSelect, "Choose the alarm sound")
soundSelect = Select("alarm-sound", [sound1, sound2, sound3, sound4, sound5], "Select a sound", { required: true })
sound1 = SelectItem("radar", "Radar (Default)")
sound2 = SelectItem("beacon", "Beacon")
sound3 = SelectItem("chimes", "Chimes")
sound4 = SelectItem("electronic", "Electronic")
sound5 = SelectItem("presto", "Presto")
