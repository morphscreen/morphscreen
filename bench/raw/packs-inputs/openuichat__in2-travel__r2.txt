root = Card([header, flightForm])
header = CardHeader("Flight Search", "Find and book your next trip")
flightForm = Form("flight-search", formButtons, [fcOrigin, fcDestination, fcTravellers, fcDates, fcCabin])
formButtons = Buttons([btnSearch, btnClear], "row")
btnSearch = Button("Search Flights", { type: "continue_conversation", context: "Search flights with the provided criteria" }, "primary")
btnClear = Button("Clear", { type: "continue_conversation", context: "Clear the flight search form" }, "tertiary")
fcOrigin = FormControl("Departure Airport", originInput, "Enter city or airport code (e.g., JFK, LAX, London)")
originInput = Input("origin", "Departure airport or city", "text", { required: true })
fcDestination = FormControl("Destination Airport", destinationInput, "Enter city or airport code (e.g., CDG, NRT, Paris)")
destinationInput = Input("destination", "Destination airport or city", "text", { required: true })
fcTravellers = FormControl("Travellers", travellersSelect, "Select number of passengers")
travellersSelect = Select("travellers", [tr1, tr2, tr3, tr4, tr5], "1 Adult", { required: true })
tr1 = SelectItem("1", "1 Adult")
tr2 = SelectItem("2", "2 Adults")
tr3 = SelectItem("3", "3 Adults")
tr4 = SelectItem("4", "4 Adults")
tr5 = SelectItem("5", "5+ Adults")
fcDates = FormControl("Travel Dates", datesPicker, "Select departure and return dates")
datesPicker = DatePicker("travel-dates", "range", { required: true })
fcCabin = FormControl("Cabin Class", cabinSelect, "Choose your preferred cabin")
cabinSelect = Select("cabin", [cb1, cb2, cb3, cb4], "Economy", {})
cb1 = SelectItem("economy", "Economy")
cb2 = SelectItem("premium", "Premium Economy")
cb3 = SelectItem("business", "Business")
cb4 = SelectItem("first", "First Class")
