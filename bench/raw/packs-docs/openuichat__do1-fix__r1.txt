bugDesc = MarkDownRenderer("**Issue:** Under concurrent load, `TokenCache.get()` can return a stale/expired token because the refresh check and token assignment are not atomic.\n\n**Root Cause:** The `if (token.isExpired())` check and subsequent `refreshToken()` call are separated by an `await`. Between the check and the refresh completing, other callers see the old expired token.\n\n**Impact:** 401 errors on downstream API calls during token rotation windows (~50-200ms). Observed in production at ~2% request rate during peak.")
bugCode = CodeBlock("typescript", "// src/auth/TokenCache.ts (before)\nclass TokenCache {\n  private token: Token | null = null;\n\n  async get(): Promise<string> {\n    if (!this.token || this.token.isExpired()) {\n      // RACE: another caller can read stale token here\n      this.token = await this.refreshToken();\n    }\n    return this.token.value;\n  }\n\n  private async refreshToken(): Promise<Token> {\n    const resp = await fetch('/auth/refresh', { method: 'POST' });\n    return resp.json();\n  }\n}")
