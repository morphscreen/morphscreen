codeBlock = CodeBlock("python", "import statistics\nimport math\n\ndef std_dev_manual(data, sample=True):\n    \"\"\"Compute standard deviation manually.\"\"\"\n    n = len(data)\n    if n < 2:\n        raise ValueError(\"Need at least 2 data points\")\n    mean = sum(data) / n\n    # Sum of squared deviations\n    ssq = sum((x - mean) ** 2 for x in data)\n    # Bessel's correction for sample\n    divisor = n - 1 if sample else n\n    variance = ssq / divisor\n    return math.sqrt(variance)\n\n# Example usage\nvalues = [12, 15, 14, 10, 18, 20, 13, 17]\n\n# Population std dev (denominator = N)\npop_sd = std_dev_manual(values, sample=False)\nprint(f\"Population σ: {pop_sd:.3f}\")\n\n# Sample std dev (denominator = n-1)\nsample_sd = std_dev_manual(values, sample=True)\nprint(f\"Sample s: {sample_sd:.3f}\")\n\n# Using standard library (recommended for production)\nprint(f\"statistics.pstdev: {statistics.pstdev(values):.3f}\")\nprint(f\"statistics.stdev:  {statistics.stdev(values):.3f}\")")
