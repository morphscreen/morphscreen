-- Create preferences table
+CREATE TABLE user_preferences (
+    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
+    theme VARCHAR(20) NOT NULL DEFAULT 'system',
+    language VARCHAR(10) NOT NULL DEFAULT 'en',
+    timezone VARCHAR(50) NOT NULL DEFAULT 'UTC',
+    email_notifications BOOLEAN NOT NULL DEFAULT true,
+    push_notifications BOOLEAN NOT NULL DEFAULT true,
+    weekly_digest BOOLEAN NOT NULL DEFAULT false,
+    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
+    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
+    PRIMARY KEY (user_id)
+);

+CREATE INDEX idx_user_preferences_theme ON user_preferences(theme);
+CREATE INDEX idx_user_preferences_language ON user_preferences(language);

-- Add trigger for updated_at
+CREATE OR REPLACE FUNCTION update_updated_at_column()
+RETURNS TRIGGER AS $$
+BEGIN
+    NEW.updated_at = NOW();
+    RETURN NEW;
+END;
+$$ language 'plpgsql';

+CREATE TRIGGER update_user_preferences_updated_at
+    BEFORE UPDATE ON user_preferences
+    FOR EACH ROW
+    EXECUTE FUNCTION update_updated_at_column();

-- Backfill existing users with default preferences
+INSERT INTO user_preferences (user_id)
+SELECT id FROM users
+ON CONFLICT (user_id) DO NOTHING;

-- Add foreign key to users table for quick preference access
+ALTER TABLE users ADD COLUMN IF NOT EXISTS preferences_json JSONB;
+UPDATE users u
+SET preferences_json = to_jsonb(p) - 'user_id'
+FROM user_preferences p WHERE p.user_id = u.id;

<Card title="3. Rollback Steps">
<Callout tone="warning" title="Rollback window: 24 hours after deployment">
After 24 hours, the migration is considered stable. Rollback after this window requires manual data migration.
</Callout>

<Steps>


<Step title="Pause application traffic" status="todo">
Enable maintenance mode or drain connections:
```bash

```
</Step>
<Step title="Run down migration" status="todo">
Execute the rollback script:


</Step>
<Step title="Restore application code" status="todo">
Deploy the previous application version (v2.3.0) that doesn't reference the new columns/models.

<Step title="Resume traffic" status="todo">

</Step>

<Callout tone="danger" title="Data loss warning">
Rolling back **deletes all user preference data** (theme, language, notification settings). This data cannot be recovered from the down migration alone. If preferences were modified post-migration, export them first:
```sql
COPY (SELECT * FROM user_preferences) TO '/backup/user_preferences_$(date +%s).csv' CSV HEADER;
```
</Callout>
